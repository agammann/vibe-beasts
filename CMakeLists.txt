cmake_minimum_required(VERSION 3.20)
project(VibeBeasts VERSION 1.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_library(partner_core src/game.cpp)
target_include_directories(partner_core PUBLIC src)
add_executable(core_tests tests/core-tests.cpp)
target_link_libraries(core_tests PRIVATE partner_core)
add_test(NAME partner_core COMMAND core_tests)
set_tests_properties(partner_core PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
add_executable(save_file_tests src/save-file.cpp tests/save-file-tests.cpp)
add_test(NAME save_file COMMAND save_file_tests)
set_tests_properties(save_file PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
include(FetchContent)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
FetchContent_Declare(raylib GIT_REPOSITORY https://github.com/raysan5/raylib.git GIT_TAG c1ab645ca298a2801097931d1079b10ff7eb9df8)
FetchContent_MakeAvailable(raylib)
add_executable(VibeBeasts src/desktop.cpp src/save-file.cpp)
target_link_libraries(VibeBeasts PRIVATE partner_core raylib)
if(WIN32)
  target_link_libraries(VibeBeasts PRIVATE winmm gdi32 opengl32)
endif()
